if __name__ == "__main__":
    which = sys.argv[1:] or ["quick"]
    t0 = time.time()
    fns = {k: v for k, v in globals().items() if k.startswith("e") and callable(v)}
    if which == ["quick"]:
        for name in QUICK:
            fns[name]()
    else:
        for w in which:
            for k, f in fns.items():
                if k.startswith(w + "_") or k == w: f()
    print(f"[{time.time()-t0:.1f}s]")
